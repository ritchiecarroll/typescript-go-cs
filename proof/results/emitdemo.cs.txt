function dist(a, b) {
    const dx = a.x - b.x, dy = a.y - b.y;
    return Math.sqrt(dx * dx + dy * dy);
}

        let n = 0;
        for (let i = 1; i < this.pts.length; i++)
            n += dist(this.pts[i - 1], this.pts[i]);
